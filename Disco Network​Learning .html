បណ្តុំពាក្យបញ្ជា (Commands Guide)

១. Modes & Navigation Commands (ការចូល និងផ្លាស់ប្តូរ Mode) 
User EXEC Mode                > Router>
Privileged EXEC Mode          > Router#
Global Config Mode            > Router(config)#
Interface Config Mode         > Router(config-if)#
Router Config Mode            > Router(config-router)#
Line Config Mode              > Router(config-line)#

enable                        # ចូលទៅ Privileged EXEC Mode (User -> Privileged)
disable                       # ថយទៅ User EXEC Mode វិញ
configure terminal            # ចូល Global Configuration Mode (conf t)
exit                          # ថយក្រោយ ១ Level
end                           # ត្រឡប់ទៅ Privileged EXEC Mode វិញភ្លាមៗ (CTRL + Z)
?                             # បង្ហាញ Help (ពាក្យបញ្ជាទាំងអស់ដែលអាចវាយបានក្នុង Mode នោះ)
show ?                        # មើលពាក្យបញ្ជា Show ទាំងអស់
២. Basic Device Setup (ការកំណត់មូលដ្ឋាន) 
hostname <NAME>               # កំណត់ឈ្មោះឧបករណ៍
banner motd ^C <MESSAGE> ^C   # បង្កើតសារព្រមានមុនពេល Login
clock set hh:mm:ss DD Month YYYY # កំណត់ម៉ោងលើ Device
service password-encryption   # កូដនីយកម្ម Password ទាំងអស់កុំឲ្យឃើញជា Plaintext
no ip domain-lookup           # បិទការ Search DNS ពេលវាយពាក្យបញ្ជាខុស

# Password Configurations
enable secret <PASSWORD>      # ដាក់ Password ចូល Enable Mode (Encrypted)
line console 0                # ចូលទៅកំណត់ Console Port
 password <PASSWORD>          # ដាក់ Password លើ Console
 login                        # បើកការទាមទារ Password
 exit

# Remote Access Setup (SSH / Telnet)
ip domain-name <DOMAIN>       # កំណត់ Domain Name សម្រាប់ SSH
crypto key generate rsa       # បង្កើត Encryption Key សម្រាប់ SSH (2048 bit)
ip ssh version 2              # ប្រើ SSH Version 2
username <USER> privilege 15 secret <PASS> # បង្កើត Admin User
line vty 0 15                 # ចូល Line Remote (0 ដល់ 15)
 login local                  # ប្រើ User local ដែលបានបង្កើត
 transport input ssh          # អនុញ្ញាតតែ SSH (ឬ all / telnet)
 exit
៣. Interface Commands (ការកំណត់ Port & IP) 
interface <INT_TYPE/NUM>      # ចូល Interface (ឧទាហរណ៍ int g0/0/0)
 interface range g0/0 - 24    # ចូល Interface ច្រើនក្នុងពេលតែមួយ
 ip address <IP> <MASK>       # ដាក់ IP Address
 ipv6 address <IPV6/PREFIX>   # ដាក់ IPv6 Address
 description <TEXT>           # ដាក់ការពិពណ៌នាលើ Port
 no shutdown                  # បើកដំណើរការ Port
 shutdown                     # បិទ Port
 duplex [auto | full | half]  # កំណត់ Duplex Mode
 speed [auto | 10 | 100 | 1000] # កំណត់ ល្បឿន Port
៤. Switching & VLAN Commands (L2 / L3 Switching) 
# VLAN Configurations
vlan <VLAN_ID>                # បង្កើត VLAN (ឧ. vlan 10)
 name <VLAN_NAME>             # ដាក់ឈ្មោះ VLAN
vtp mode [server | client | transparent | off] # កំណត់ VTP Mode
vtp domain <DOMAIN_NAME>      # កំណត់ VTP Domain

# Switchport Modes
interface <INT>
 switchport mode access       # កំណត់ Mode ជា Access
 switchport access vlan <ID>  # ដាក់ Port ចូលក្នុង VLAN
 switchport mode trunk        # កំណត់ Mode ជា Trunk
 switchport trunk allowed vlan <IDs> # កំណត់ VLAN ដែលអនុញ្ញាតលើ Trunk
 switchport trunk native vlan <ID>   # កំណត់ Native VLAN

# Spanning Tree Protocol (STP)
spanning-tree mode [pvst | rapid-pvst | mst] # កំណត់ STP Mode
spanning-tree vlan <ID> root primary        # កំណត់ Switch ជា Primary Root Bridge
spanning-tree portfast                      # បើក PortFast លើ Access Port
spanning-tree bpduguard enable              # បើក BPDU Guard ការពារ Loop

# EtherChannel (Link Aggregation)
interface range g0/1 - 2
 channel-group 1 mode [active | passive | auto | desirable | on] # បង្កើត EtherChannel
interface port-channel 1
 switchport mode trunk
៥. IP Services & DHCP Commands 
# DHCP Server Setup
ip dhcp excluded-address <START_IP> <END_IP> # លុប IP មិនឲ្យចែក
ip dhcp pool <POOL_NAME>      # បង្កើត DHCP Pool
 network <NET_IP> <MASK>      # កំណត់ Network
 default-router <GW_IP>       # កំណត់ Default Gateway
 dns-server <DNS1> <DNS2>     # កំណត់ DNS Server
 domain-name <DOMAIN>         # កំណត់ Domain Name
 lease <DAYS> <HOURS> <MINS>  # កំណត់ Lease Time
 exit

# DHCP Relay Agent (លើ Router Interface)
interface <INT>
 ip helper-address <DHCP_SERVER_IP> # ចម្លង DHCP Request ទៅ DHCP Server ឆ្ងាយ
 exit
៦. Routing Commands (Static & Dynamic Protocols) 
# Static & Default Routes
ip route <DEST_NET> <MASK> <NEXT_HOP_IP> # Static Route
ip route 0.0.0.0 0.0.0.0 <NEXT_HOP_IP>   # Default Route

# RIP Protocol
router rip
 version 2
 network <NET_IP>             # ប្រកាស Classful Network
 no auto-summary              # បិទ Auto Summary
 exit

# OSPF Protocol
router ospf <PROCESS_ID>
 router-id <ID_IP>            # កំណត់ Router ID (ឧ. 1.1.1.1)
 network <NET_IP> <WILDCARD> area <AREA_ID> # ប្រកាស Network ជាមួយ Wildcard
 passive-interface <INT>      # បិទ OSPF Hello លើ Port Client
 exit

# EIGRP Protocol
router eigrp <AS_NUMBER>
 router-id <ID_IP>
 network <NET_IP> <WILDCARD>
 no auto-summary
 exit

# BGP Protocol
router bgp <AS_NUMBER>
 neighbor <IP> remote-as <REMOTE_AS> # ភ្ជាប់ជាមួយ BGP Neighbor
 network <NET_IP> mask <MASK>        # ប្រកាស Network
 exit
៧. Security & Access Control Lists (ACL) 
# Standard ACL (1-99, 1300-1999) - ពិនិត្យតែ Source IP
access-list <1-99> [permit | deny] <SOURCE_IP> <WILDCARD>
access-list 1 permit host 192.168.1.10

# Extended ACL (100-199, 2000-2699) - ពិនិត្យ Src, Dst, Protocol & Port
access-list <100-199> [permit | deny] <PROTOCOL> <SRC_IP> <SRC_WILD> <DST_IP> <DST_WILD> eq <PORT>
access-list 100 deny tcp host 192.168.1.50 host 10.0.0.1 eq 80
access-list 100 permit ip any any

# Named ACL (Standard & Extended)
ip access-list extended <NAME>
 deny tcp host 192.168.1.50 host 10.0.0.1 eq 443
 permit ip any any
 exit

# Apply ACL ទៅ Interface ឬ Line
interface <INT>
 ip access-group <ACL_NAME/NUM> [in | out]
line vty 0 15
 access-class <ACL_NAME/NUM> in               # កំណត់ IP ដែលអាច SSH/Telnet ចូលបាន

# Switch Port Security
interface <INT>
 switchport mode access
 switchport port-security
 switchport port-security maximum <NUM>
 switchport port-security mac-address sticky
 switchport port-security violation [protect | restrict | shutdown]
៨. NAT / PAT Commands (Network Address Translation) 
# 1. កំណត់ Inside & Outside Interfaces
interface <INT_INSIDE>
 ip nat inside
interface <INT_OUTSIDE>
 ip nat outside

# Static NAT (1 to 1 Mapping)
ip nat inside source static <PRIVATE_IP> <PUBLIC_IP>

# Dynamic NAT / PAT (Port Address Translation - Overload)
access-list 1 permit 192.168.1.0 0.0.0.255
ip nat inside source list 1 interface <OUTSIDE_INT> overload
៩. Troubleshooting & Monitoring Commands (Show & Debug) 
# Verification (Show Commands)
show running-config           # មើល Running Configuration ទាំងអស់ (sh run)
show startup-config           # មើល Configuration ដែលបាន Save ក្នុង NVRAM
show ip interface brief       # មើល Interface Status & IP (sh ip int br)
show ip route                 # មើល Routing Table
show vlan brief               # មើល VLANs ទាំងអស់
show interfaces status        # មើល Status នៃ Switch Ports
show mac address-table        # មើល MAC Address Table លើ Switch
show cdp neighbors            # មើល Cisco Devices ដែលភ្ជាប់ជាប់គ្នា (sh cdp nei)
show cdp neighbors detail     # មើល Detail/IP របស់ Neighbor Devices
show ip dhcp binding          # មើល IP ដែល DHCP បានចែក
show ip nat translations      # មើល តារាង NAT Translation
show access-lists             # មើល ACL និង Match Counts
show processes cpu            # មើល CPU Utilization
show memory statistics        # មើល RAM Utilization
show log                      # មើល System Logs

# Diagnostic Tools
ping <IP>                     # Test Connectivity
traceroute <IP>               # ស្វែងរក Path ទៅ Destination
debug ip packet               # មើល IP Packets ក្នុង Real-time
debug ip ospf events          # Debug OSPF
undebug all                   # បិទ Debug ទាំងអស់ (un all)
១០. File System & Maintenance Commands 
copy running-config startup-config  # រក្សាទុក Config (ឬវាយ: write / wr)
copy running-config tftp:          # Backup Config ទៅ TFTP Server
copy tftp: running-config          # Restore Config ពី TFTP Server
copy flash: tftp:                  # Backup IOS Image
erase startup-config               # លុប Config ទាំងអស់ក្នុង NVRAM (Reset Device)
delete flash:<FILENAME>            # លុប File ក្នុង Flash Memory
dir flash:                         # មើល File ទាំងអស់ក្នុង Flash Memory
reload                             # Reboot ឧបករណ៍
